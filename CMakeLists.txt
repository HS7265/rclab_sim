cmake_minimum_required(VERSION 3.22)
project(rclab_sim VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RCLAB_BUILD_GUI "Build GLFW/ImGui/ImPlot viewer" ON)
option(RCLAB_WITH_ONNX "Build ONNX Runtime robot control stacks" OFF)
set(MUJOCO_ROOT "$ENV{MUJOCO_PATH}" CACHE PATH "MuJoCo 3.11.0 installation")
set(RCLAB_EXTENSION_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/examples/sine_controller.cpp" CACHE STRING "Sources implementing rclab::register_extensions()")
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include" REQUIRED)
find_library(MUJOCO_LIBRARY mujoco HINTS "${MUJOCO_ROOT}/lib" REQUIRED)
# 일부 로컬 SDK는 SONAME 심링크가 빠져 있다. 원본 설치 대신 빌드 디렉터리에 보완한다.
if(UNIX AND NOT APPLE)
  find_program(READELF_EXECUTABLE readelf REQUIRED)
  execute_process(COMMAND ${READELF_EXECUTABLE} -d ${MUJOCO_LIBRARY} OUTPUT_VARIABLE MUJOCO_ELF)
  string(REGEX MATCH "SONAME[^\n]*\\[([^]]+)\\]" MUJOCO_SONAME_MATCH "${MUJOCO_ELF}")
  if(CMAKE_MATCH_1)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/mujoco-runtime")
    file(CREATE_LINK "${MUJOCO_LIBRARY}" "${CMAKE_BINARY_DIR}/mujoco-runtime/${CMAKE_MATCH_1}" SYMBOLIC)
    list(APPEND CMAKE_BUILD_RPATH "${CMAKE_BINARY_DIR}/mujoco-runtime")
  endif()
endif()
find_package(Threads REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY zstd REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(rclab_core src/profile.cpp src/simulation.cpp src/recording.cpp src/runner.cpp src/builtin_stacks.cpp)
target_include_directories(rclab_core PUBLIC include "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(rclab_core PUBLIC "${MUJOCO_LIBRARY}" nlohmann_json::nlohmann_json Threads::Threads)
target_include_directories(rclab_core PRIVATE "${ZSTD_INCLUDE_DIR}")
target_link_libraries(rclab_core PRIVATE "${ZSTD_LIBRARY}")
target_compile_options(rclab_core PRIVATE -Wall -Wextra -Wpedantic)
if(RCLAB_WITH_ONNX)
  set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C++ SDK root")
  find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
  find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
  find_package(OpenSSL REQUIRED)
  target_sources(rclab_core PRIVATE src/rok4_student.cpp)
  target_include_directories(rclab_core PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
  target_link_libraries(rclab_core PUBLIC "${ONNXRUNTIME_LIBRARY}" OpenSSL::Crypto)
  target_compile_definitions(rclab_core PRIVATE RCLAB_WITH_ONNX)
endif()
add_executable(rclab_headless src/headless.cpp ${RCLAB_EXTENSION_SOURCES})
target_link_libraries(rclab_headless PRIVATE rclab_core)
if(RCLAB_BUILD_GUI)
  include(FetchContent)
  find_package(glfw3 3.3 REQUIRED)
  find_package(OpenGL REQUIRED)
  FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.89.9.tar.gz)
  FetchContent_Declare(implot URL https://github.com/epezent/implot/archive/refs/tags/v0.16.tar.gz)
  FetchContent_GetProperties(imgui)
  if(NOT imgui_POPULATED)
    FetchContent_Populate(imgui)
  endif()
  FetchContent_GetProperties(implot)
  if(NOT implot_POPULATED)
    FetchContent_Populate(implot)
  endif()
  add_library(rclab_ui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
    ${implot_SOURCE_DIR}/implot.cpp ${implot_SOURCE_DIR}/implot_items.cpp)
  target_include_directories(rclab_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${implot_SOURCE_DIR})
  target_link_libraries(rclab_ui PUBLIC glfw OpenGL::GL ${CMAKE_DL_LIBS})
  add_executable(rclab_sim src/gui.cpp ${RCLAB_EXTENSION_SOURCES})
  target_link_libraries(rclab_sim PRIVATE rclab_core rclab_ui)
endif()
include(CTest)
if(BUILD_TESTING)
  set(RCLAB_ROK4_REFERENCE_ROOT "" CACHE PATH "Optional external ROK4 reference C++ source root")
  set(RCLAB_ROK4_STUDENT_PROFILE "" CACHE FILEPATH "Optional local student profile")
  set(RCLAB_ROK4_STUDENT_BUNDLE "" CACHE PATH "Optional local student bundle")
  if(RCLAB_WITH_ONNX AND RCLAB_ROK4_REFERENCE_ROOT AND RCLAB_ROK4_STUDENT_PROFILE AND RCLAB_ROK4_STUDENT_BUNDLE)
    add_executable(rclab_student_tests tests/rok4_student_tests.cpp
      ${RCLAB_ROK4_REFERENCE_ROOT}/src/rok4/rok4_core.cpp)
    target_include_directories(rclab_student_tests PRIVATE ${RCLAB_ROK4_REFERENCE_ROOT}/include ${ONNXRUNTIME_INCLUDE_DIR})
    target_link_libraries(rclab_student_tests PRIVATE rclab_core)
    add_test(NAME student_reference COMMAND rclab_student_tests ${RCLAB_ROK4_STUDENT_PROFILE} ${RCLAB_ROK4_STUDENT_BUNDLE})
  endif()
  add_executable(rclab_tests tests/core_tests.cpp)
  target_link_libraries(rclab_tests PRIVATE rclab_core)
  target_compile_definitions(rclab_tests PRIVATE RCLAB_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME core COMMAND rclab_tests)
  foreach(robot biped quadruped hexapod humanoid)
    add_test(NAME model_${robot} COMMAND rclab_headless --robot ${CMAKE_CURRENT_SOURCE_DIR}/profiles/${robot}.json --duration 0.2)
  endforeach()
  set(RCLAB_RUBI_MODEL "" CACHE FILEPATH "Optional external RUBI MJCF")
  set(RCLAB_ROK4_MODEL "" CACHE FILEPATH "Optional external ROK-4 MJCF")
  foreach(robot RUBI ROK4)
    if(RCLAB_${robot}_MODEL)
      string(TOLOWER "${robot}" robot_lower)
      add_test(NAME external_${robot_lower} COMMAND rclab_tests --external ${CMAKE_CURRENT_SOURCE_DIR}/profiles/${robot_lower}.example.json ${RCLAB_${robot}_MODEL})
    endif()
  endforeach()
endif()
